\begin{enumerate}[label=\textbf{\arabic*.}, start=18, leftmargin=*, itemsep=1.3em]
    \item Đẳng thức $f \circ (g + h) = f \circ g + f \circ h$ có đúng không?

    \item Chứng minh rằng nếu $n$ là một số nguyên dương thì $7^n - 1$ chia hết cho $6$.

    \item Chứng minh rằng $1 + 3 + 5 + \dots + (2n - 1) = n^2$.

    \item Cho $f_0(x) = x^2$ và $f_{n+1}(x) = f_0(f_n(x))$ với $n = 0, 1, 2, \dots$, hãy tìm công thức tính $f_n(x)$.

    \item 
    \begin{enumerate}[label=(\alph*), leftmargin=*, itemsep=0.8em]
        \item Cho $f_0(x) = \dfrac{1}{2 - x}$ và $f_{n+1} = f_0 \circ f_n$ với $n = 0, 1, 2, \dots$, hãy tìm một biểu thức biểu diễn $f_n(x)$ và dùng phương pháp quy nạp toán học để chứng minh điều đó.
        \item \llap{\raisebox{-0.15ex}{\includegraphics[height=1.05em]{images/p0111_fig1.png}}\hspace{0.45em}}Vẽ đồ thị của $f_0, f_1, f_2, f_3$ trên cùng một màn hình và mô tả các hiệu ứng của phép hợp hàm liên tiếp.
    \end{enumerate}
\end{enumerate}
