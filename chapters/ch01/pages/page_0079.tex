\noindent\textbf{\textsf{\color{stewartcyan}59.}}\quad Một hòn đá được thả xuống hồ, tạo ra một gợn sóng hình tròn lan rộng ra ngoài với tốc độ $60\text{ cm/s}$.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=1pt,topsep=2pt]
    \item Biểu diễn bán kính $r$ của hình tròn theo thời gian $t$ (tính bằng giây).
    \item Nếu $A$ là diện tích của hình tròn theo bán kính, hãy tìm $A \circ r$ và nêu ý nghĩa của nó.
\end{enumerate}

\vspace{1pt}
\noindent\textbf{\textsf{\color{stewartcyan}60.}}\quad Một quả bóng hình cầu đang được bơm căng và bán kính của nó tăng với tốc độ $2\text{ cm/s}$.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=1pt,topsep=2pt]
    \item Biểu diễn bán kính $r$ của quả bóng theo thời gian $t$ (tính bằng giây).
    \item Nếu $V$ là thể tích của quả bóng theo bán kính, hãy tìm $V \circ r$ và nêu ý nghĩa của nó.
\end{enumerate}

\vspace{1pt}
\noindent\textbf{\textsf{\color{stewartcyan}61.}}\quad Một con tàu đang di chuyển với vận tốc $30\text{ km/h}$ song song với một bờ biển thẳng. Con tàu cách bờ $6\text{ km}$ và đi ngang qua một ngọn hải đăng vào đúng buổi trưa.
\begin{enumerate}[label=(\alph*),leftmargin=*,itemsep=1pt,topsep=2pt]
    \item Biểu diễn khoảng cách $s$ giữa ngọn hải đăng và con tàu theo $d$, với $d$ là quãng đường tàu đã đi được kể từ buổi trưa; nghĩa là tìm hàm số $f$ sao cho $s = f(d)$.
    \item Biểu diễn $d$ theo thời gian $t$ đã trôi qua kể từ buổi trưa; nghĩa là tìm hàm số $g$ sao cho $d = g(t)$.
    \item Tìm $f \circ g$. Hàm số này biểu thị đại lượng nào?
\end{enumerate}
