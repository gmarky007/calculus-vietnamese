    \item Cho $f$ là hàm số có đồ thị như trong hình.
    {\centering\vspace{1pt}\includegraphics[width=0.52\linewidth]{images/p0103_fig1.png}\par\vspace{1pt}}
    \begin{enumerate}[label=(\alph*), itemsep=0.5pt, topsep=1pt, parsep=0pt]
        \item Ước lượng giá trị của $f(2)$.
        \item Ước lượng các giá trị của $x$ sao cho $f(x) = 3$.
        \item Nêu tập xác định của $f$.
        \item Nêu tập giá trị của $f$.
        \item Hàm số $f$ đồng biến trên khoảng nào?
        \item $f$ có phải là hàm đơn ánh không? Giải thích.
        \item $f$ là hàm chẵn, lẻ hay không chẵn cũng không lẻ? Giải thích.
    \end{enumerate}

    \item Cho đồ thị của $g$.
    {\centering\vspace{1pt}\includegraphics[width=0.45\linewidth]{images/p0103_fig3.png}\par\vspace{1pt}}
    \begin{enumerate}[label=(\alph*), itemsep=0.5pt, topsep=1pt, parsep=0pt]
        \item Nêu giá trị của $g(2)$.
        \item Vì sao $g$ là hàm đơn ánh?
        \item Ước lượng giá trị của $g^{-1}(2)$.
        \item Ước lượng tập xác định của $g^{-1}$.
        \item Vẽ phác thảo đồ thị của $g^{-1}$.
    \end{enumerate}

    \item Nếu $f(x) = x^2 - 2x + 3$, hãy tính tỉ sai phân
    \setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{2pt}
    \[
    \frac{f(a+h) - f(a)}{h}
    \]

    \item Hãy vẽ phác thảo đồ thị năng suất cây trồng như một hàm số theo lượng phân bón sử dụng.
\end{enumerate}

    \begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}ll@{}}
    (a) $f(x) = 2x^5 - 3x^2 + 2$ & (b) $f(x) = x^3 - x^7$ \\[1pt]
    (c) $f(x) = e^{-x^2}$ & (d) $f(x) = 1 + \sin x$ \\[1pt]
    (e) $f(x) = 1 - \cos 2x$ & (f) $f(x) = (x + 1)^2$

    \item Tìm công thức xác định hàm số có đồ thị gồm đoạn thẳng nối từ điểm $(-2, 2)$ đến điểm $(-1, 0)$ kết hợp với nửa trên của đường tròn tâm tại gốc tọa độ và bán kính bằng $1$.

    \item Nếu $f(x) = \ln x$ và $g(x) = x^2 - 9$, hãy tìm các hàm số: (a) $f \circ g$, (b) $g \circ f$, (c) $f \circ f$, (d) $g \circ g$, và tìm tập xác định của chúng.

    \item Hãy biểu diễn hàm số $F(x) = 1/\sqrt{x + \sqrt{x}}$ dưới dạng hợp của ba hàm số.

Hãy sử dụng biểu đồ phân tán để chọn loại mô hình thích hợp. Sử dụng mô hình của bạn để dự đoán tuổi thọ của một bé trai sinh vào năm 2030.
