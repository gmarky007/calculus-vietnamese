    \noindent\textbf{69.} Tại những điểm nào trên đường cong $y = \sin x + \cos x$, $0 \leqslant x \leqslant 2\pi$, thì tiếp tuyến nằm ngang?

    \par\medskip
    \noindent\textbf{70.} Tìm các điểm trên elip $x^2 + 2y^2 = 1$ mà tại đó tiếp tuyến có hệ số góc bằng 1.

    \par\medskip
    \noindent\textbf{71.} Nếu $f(x) = (x - a)(x - b)(x - c)$, hãy chứng minh rằng
    \[
    \frac{f'(x)}{f(x)} = \frac{1}{x - a} + \frac{1}{x - b} + \frac{1}{x - c}
    \]

    \noindent\textbf{72.} \textbf{(a)} Bằng cách lấy đạo hàm công thức góc nhân đôi
    \[
    \cos 2x = \cos^2 x - \sin^2 x
    \]
    hãy suy ra công thức góc nhân đôi cho hàm số sin.
    \par\smallskip
    \noindent\hspace*{1.4em}\textbf{(b)} Bằng cách lấy đạo hàm công thức cộng
    \[
    \sin(x + a) = \sin x \cos a + \cos x \sin a
    \]
    hãy suy ra công thức cộng cho hàm số côsin.

    \par\medskip
    \noindent\textbf{73.} Giả sử rằng
    \[
    \begin{array}{llll}
    f(1) = 2 & f'(1) = 3 & f(2) = 1 & f'(2) = 2 \\
    g(1) = 3 & g'(1) = 1 & g(2) = 1 & g'(2) = 4
    \end{array}
    \]
    \textbf{(a)} Nếu $S(x) = f(x) + g(x)$, hãy tìm $S'(1)$.\\
    \textbf{(b)} Nếu $P(x) = f(x)g(x)$, hãy tìm $P'(2)$.\\
    \textbf{(c)} Nếu $Q(x) = f(x)/g(x)$, hãy tìm $Q'(1)$.\\
    \textbf{(d)} Nếu $C(x) = f(g(x))$, hãy tìm $C'(2)$.

    \par\medskip
    \noindent\textbf{74.} Nếu $f$ và $g$ là các hàm số có đồ thị như hình vẽ, đặt $P(x) = f(x)g(x)$, $Q(x) = f(x)/g(x)$ và $C(x) = f(g(x))$. Hãy tìm \textbf{(a)} $P'(2)$, \textbf{(b)} $Q'(2)$ và \textbf{(c)} $C'(2)$.

    \par\smallskip

    \noindent\textbf{89.} Tìm parabol $y = ax^2 + bx + c$ đi qua điểm $(1, 4)$ và có các tiếp tuyến tại $x = -1$ và $x = 5$ lần lượt có hệ số góc bằng $6$ và $-2$.

    \noindent\textbf{90.} Hàm số $C(t) = K(e^{-at} - e^{-bt})$, trong đó $a, b$ và $K$ là các hằng số dương và $b > a$, được dùng để mô hình hóa nồng độ tại thời điểm $t$ của một loại thuốc được tiêm vào máu.

    \noindent\textbf{92.} Một chất điểm chuyển động dọc theo một đường thẳng nằm ngang sao cho tọa độ của nó tại thời điểm $t$ là $x = \sqrt{b^2 + c^2 t^2}$, $t \geqslant 0$, trong đó $b$ và $c$ là các hằng số dương.
