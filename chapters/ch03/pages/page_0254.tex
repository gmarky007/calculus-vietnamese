  \noindent\textbf{\textsf{\color{stewartcyan}VÍ DỤ 4}}\quad Tính đạo hàm của $f(x) = \log_{10}(2 + \sin x)$.

  \vspace{5pt}
  \noindent\textbf{\textsf{\color{stewartcyan}LỜI GIẢI}}\quad Áp dụng Công thức~1 với $b = 10$, ta có
  \begin{align*}
    f'(x) &= \frac{d}{dx} \log_{10}(2 + \sin x) \\[3pt]
    &= \frac{1}{(2 + \sin x) \ln 10} \frac{d}{dx} (2 + \sin x) \\[3pt]
    &= \frac{\cos x}{(2 + \sin x) \ln 10} \tag*{\color{stewartcyan}\blacksquare}
  \end{align*}

  \vspace{10pt}
  \noindent\textbf{\textsf{\color{stewartcyan}VÍ DỤ 5}}\quad Tìm $\displaystyle \frac{d}{dx} \ln \frac{x + 1}{\sqrt{x - 2}}$.

  \vspace{6pt}
  \noindent\textbf{\textsf{\color{stewartcyan}LỜI GIẢI 1}}
  \begin{align*}
    \frac{d}{dx} \ln \frac{x + 1}{\sqrt{x - 2}} &= \frac{1}{\dfrac{x + 1}{\sqrt{x - 2}}} \frac{d}{dx} \frac{x + 1}{\sqrt{x - 2}} \\[4pt]
    &= \frac{\sqrt{x - 2}}{x + 1} \cdot \frac{\sqrt{x - 2}\cdot 1 - (x + 1)\left(\frac{1}{2}\right)(x - 2)^{-1/2}}{x - 2} \\[4pt]
    &= \frac{x - 2 - \frac{1}{2}(x + 1)}{(x + 1)(x - 2)} \\[4pt]
    &= \frac{x - 5}{2(x + 1)(x - 2)}
  \end{align*}

  \vspace{6pt}
  \noindent\textbf{\textsf{\color{stewartcyan}LỜI GIẢI 2}}\quad Nếu trước hết ta khai triển hàm số đã cho bằng cách dùng các quy tắc logarit, thì việc lấy đạo hàm sẽ trở nên dễ dàng hơn:
  \begin{align*}
    \frac{d}{dx} \ln \frac{x + 1}{\sqrt{x - 2}} &= \frac{d}{dx} \left[ \ln(x + 1) - \tfrac{1}{2}\ln(x - 2) \right] \\[3pt]
    &= \frac{1}{x + 1} - \frac{1}{2}\left(\frac{1}{x - 2}\right)
  \end{align*}
  (Kết quả này có thể để nguyên như vậy, nhưng nếu ta quy đồng mẫu số thì sẽ thấy nó cho cùng kết quả như ở Lời giải~1.) \hfill {\color{stewartcyan}$\blacksquare$}

  \vspace{12pt}
  \noindent\textbf{\textsf{\color{stewartcyan}VÍ DỤ 6}}\quad Tìm $f'(x)$ nếu $f(x) = \ln|x|$.

  \vspace{5pt}
  \noindent\textbf{\textsf{\color{stewartcyan}LỜI GIẢI}}\quad Vì
  \[
    f(x) = \begin{cases}
      \ln x & \text{nếu } x > 0 \\[2pt]
      \ln(-x) & \text{nếu } x < 0
    \end{cases}
  \]
  nên suy ra
  \[
    f'(x) = \begin{cases}
      \dfrac{1}{x} & \text{nếu } x > 0 \\[10pt]
      \dfrac{1}{-x}(-1) = \dfrac{1}{x} & \text{nếu } x < 0
    \end{cases}
  \]
  Do đó $f'(x) = 1/x$ với mọi $x \ne 0$. \hfill {\color{stewartcyan}$\blacksquare$}
