    \begin{equation*}
        \frac{dy}{dx} = -\frac{x}{y} = -\frac{x}{-\sqrt{25 - x^2}} = \frac{x}{\sqrt{25 - x^2}}

    \vspace{0.15cm}
    \noindent
    \textbf{\textsf{\color{stewartcyan}VÍ DỤ 2}}\quad
    \textbf{(a)} Tìm $y'$ nếu $x^3 + y^3 = 6xy$.\par
    \noindent
    \textbf{(b)} Tìm tiếp tuyến của lá Descartes $x^3 + y^3 = 6xy$ tại điểm $(3, 3)$.\par
    \noindent
    \textbf{(c)} Tại điểm nào trong góc phần tư thứ nhất thì tiếp tuyến nằm ngang?

    \vspace{0.2cm}
    \noindent
    \textbf{\textsf{\color{stewartcyan}LỜI GIẢI}}\par
    \noindent
    \textbf{(a)} Lấy đạo hàm hai vế của phương trình $x^3 + y^3 = 6xy$ theo biến $x$, xem $y$ là một hàm số của $x$, đồng thời áp dụng Quy tắc chuỗi cho số hạng $y^3$ và Quy tắc tích cho số hạng $6xy$, ta được
    \begin{align*}
        3x^2 + 3y^2y' &= 6xy' + 6y \\
        \shortintertext{hay}
        x^2 + y^2y' &= 2xy' + 2y
    \end{align*}
    \noindent
    \begin{minipage}[t]{0.35\linewidth}
        \vspace{0pt}
        Bây giờ ta giải tìm $y'$:
    \end{minipage}%
    \begin{minipage}[t]{0.65\linewidth}
        \vspace{-1.2\baselineskip}
        \begin{align*}
            y^2y' - 2xy' &= 2y - x^2 \\
            (y^2 - 2x)y' &= 2y - x^2 \\
            y' &= \frac{2y - x^2}{y^2 - 2x}
        \end{align*}
    \end{minipage}

    \vspace{0.1cm}
    \noindent
    \textbf{(b)} Khi $x = y = 3$,
    \begin{equation*}
        y' = \frac{2 \cdot 3 - 3^2}{3^2 - 2 \cdot 3} = -1
    \end{equation*}
    và nhìn lướt qua Hình 4 xác nhận rằng đây là một giá trị hợp lý cho hệ số góc tại $(3, 3)$. Do đó, phương trình tiếp tuyến của lá Descartes tại $(3, 3)$ là
    \begin{equation*}
        y - 3 = -1(x - 3) \qquad\text{hay}\qquad x + y = 6
    \end{equation*}

    \noindent
    \textbf{(c)} Tiếp tuyến nằm ngang nếu $y' = 0$. Sử dụng biểu thức của $y'$ ở phần (a), ta thấy rằng $y' = 0$ khi $2y - x^2 = 0$ (với điều kiện $y^2 - 2x \neq 0$). Thay $y = \frac{1}{2}x^2$ vào phương trình của đường cong, ta được
    \begin{equation*}
        x^3 + \left(\tfrac{1}{2}x^2\right)^3 = 6x\left(\tfrac{1}{2}x^2\right)
    \end{equation*}
    phương trình này rút gọn thành $x^6 = 16x^3$. Vì $x \neq 0$ trong góc phần tư thứ nhất, nên ta có $x^3 = 16$. Nếu $x = 16^{1/3} = 2^{4/3}$, thì $y = \frac{1}{2}(2^{8/3}) = 2^{5/3}$. Như vậy, tiếp tuyến nằm ngang tại $(2^{4/3}, 2^{5/3})$, xấp xỉ bằng $(2{,}5198;\, 3{,}1748)$. Nhìn vào Hình 5, ta thấy kết quả này là hợp lý.\hfill\textcolor{stewartcyan}{\rule{1.1ex}{1.1ex}}

    \vspace{0.25cm}
    \noindent
    \textbf{\textsf{\color{stewartgreen}GHI CHÚ 2}}\quad Có một công thức cho ba nghiệm của một phương trình bậc ba tương tự như công thức nghiệm của phương trình bậc hai nhưng phức tạp hơn nhiều. Nếu dùng công thức này (hoặc máy tính) để giải phương trình $x^3 + y^3 = 6xy$ tìm $y$ theo $x$, ta nhận được ba hàm số được xác định bởi các phương trình:
    \begin{equation*}
        y = f(x) = \sqrt[3]{-\tfrac{1}{2}x^3 + \sqrt{\tfrac{1}{4}x^6 - 8x^3}} + \sqrt[3]{-\tfrac{1}{2}x^3 - \sqrt{\tfrac{1}{4}x^6 - 8x^3}}
    \end{equation*}
    \noindent và
    \begin{equation*}
        y = \tfrac{1}{2}\left[-f(x) \pm \sqrt{-3}\left(\sqrt[3]{-\tfrac{1}{2}x^3 + \sqrt{\tfrac{1}{4}x^6 - 8x^3}} - \sqrt[3]{-\tfrac{1}{2}x^3 - \sqrt{\tfrac{1}{4}x^6 - 8x^3}}\right)\right]
    \end{equation*}
