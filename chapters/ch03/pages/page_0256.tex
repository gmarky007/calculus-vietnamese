\begin{minipage}[t]{\rightcolwidth}\vspace{0pt}
  \noindent\textbf{\textsf{\color{stewartcyan}VÍ DỤ 8}}\hspace{1em}Tính đạo hàm của $y = x^{\sqrt{x}}$.\par\vspace{8pt}

  \noindent\textbf{\textsf{\color{stewartcyan}LỜI GIẢI 1}}\hspace{1em}Vì cả cơ số và số mũ đều là biến, ta dùng phương pháp đạo hàm logarit:
  \begin{align*}
  \ln y &= \ln x^{\sqrt{x}} = \sqrt{x}\ln x\\[5pt]
  \frac{y'}{y} &= \sqrt{x}\cdot\frac{1}{x} + (\ln x)\frac{1}{2\sqrt{x}}\\[5pt]
  y' &= y\left(\frac{1}{\sqrt{x}} + \frac{\ln x}{2\sqrt{x}}\right) = x^{\sqrt{x}}\left(\frac{2 + \ln x}{2\sqrt{x}}\right)
  \end{align*}

  \noindent\textbf{\textsf{\color{stewartcyan}LỜI GIẢI 2}}\hspace{1em}Một phương pháp khác là sử dụng Phương trình~1.5.10 để viết $x^{\sqrt{x}} = e^{\sqrt{x}\ln x}$:
  \begin{align*}
  \frac{d}{dx}\left(x^{\sqrt{x}}\right) &= \frac{d}{dx}\left(e^{\sqrt{x}\ln x}\right) = e^{\sqrt{x}\ln x}\frac{d}{dx}\left(\sqrt{x}\ln x\right)\\[5pt]
  &= x^{\sqrt{x}}\left(\frac{2 + \ln x}{2\sqrt{x}}\right)\qquad\textsf{\color{stewartcyan}(như trong Lời giải 1)}\tag*{\color{stewartcyan}\blacksquare}
  \end{align*}

  \vspace{18pt}

  \noindent{\color{stewartcyan}\rule{2.2mm}{2.2mm}}\hspace{0.6em}{\textbf{\textsf{\large Số $e$ dưới dạng một giới hạn}}}\par\vspace{8pt}

  \noindent Chúng ta đã chỉ ra rằng nếu $f(x) = \ln x$, thì $f'(x) = 1/x$. Do đó $f'(1) = 1$. Bây giờ chúng ta sử dụng kết quả này để biểu diễn số $e$ dưới dạng một giới hạn.\par\vspace{6pt}

  \noindent Từ định nghĩa của đạo hàm dưới dạng một giới hạn, ta có
  \begin{align*}
  f'(1) &= \lim_{h \to 0}\frac{f(1 + h) - f(1)}{h} = \lim_{x \to 0}\frac{f(1 + x) - f(1)}{x}\\[6pt]
  &= \lim_{x \to 0}\frac{\ln(1 + x) - \ln 1}{x} = \lim_{x \to 0}\frac{1}{x}\ln(1 + x)\\[6pt]
  &= \lim_{x \to 0}\ln(1 + x)^{1/x}
  \end{align*}
\end{minipage}
