\textbf{40.}\enskip Phác thảo đồ thị của một hàm số $g$ thỏa mãn $g(0) = g(2) = g(4) = 0$, $g'(1) = g'(3) = 0$, $g'(0) = g'(4) = 1$, $g'(2) = -1$, $\lim_{x\to\infty} g(x) = \infty$, và $\lim_{x\to -\infty} g(x) = -\infty$.

\vspace{7pt}

\textbf{41.}\enskip Phác thảo đồ thị của một hàm số $g$ liên tục trên tập xác định $(-5, 5)$ của nó và thỏa mãn $g(0) = 1$, $g'(0) = 1$, $g'(-2) = 0$, $\lim_{x\to -5^+} g(x) = \infty$, và $\lim_{x\to 5^-} g(x) = 3$.

\vspace{7pt}

\textbf{42.}\enskip Phác thảo đồ thị của một hàm số $f$ có miền xác định là $(-2, 2)$, $f'(0) = -2$, $\lim_{x\to 2^-} f(x) = \infty$, $f$ liên tục tại mọi điểm thuộc miền xác định ngoại trừ $\pm 1$, và $f$ là hàm lẻ.

\vspace{7pt}

{\color{stewartblue}\textbf{43–48}}\enskip Mỗi giới hạn sau biểu diễn đạo hàm của một hàm số $f$ nào đó tại một số $a$. Hãy xác định hàm $f$ và số $a$ trong từng trường hợp.

\vspace{5pt}

\noindent
\begin{tabular}{@{}p{0.48\columnwidth}@{\hspace{0.04\columnwidth}}p{0.48\columnwidth}@{}}
\textbf{43.} $\displaystyle \lim_{h\to 0} \frac{\sqrt{9+h} - 3}{h}$ & \textbf{44.} $\displaystyle \lim_{h\to 0} \frac{e^{-2+h} - e^{-2}}{h}$ \\[12pt]
\textbf{45.} $\displaystyle \lim_{x\to 2} \frac{x^6 - 64}{x - 2}$ & \textbf{46.} $\displaystyle \lim_{x\to 1/4} \frac{\dfrac{1}{x} - 4}{x - \dfrac{1}{4}}$ \\[15pt]
\textbf{47.} $\displaystyle \lim_{h\to 0} \frac{\tan\left(\dfrac{\pi}{4}+h\right) - 1}{h}$ & \textbf{48.} $\displaystyle \lim_{\theta\to \pi/6} \frac{\sin\theta - \dfrac{1}{2}}{\theta - \dfrac{\pi}{6}}$
\end{tabular}

\textbf{49.}\enskip Chi phí (tính bằng đô la) để sản xuất $x$ đơn vị của một loại hàng hóa là $C(x) = 5000 + 10x + 0{,}05x^2$.
\begin{enumerate}[label=(\alph*),leftmargin=1.4em,itemsep=1pt,topsep=2pt]
  \item Tìm tốc độ thay đổi trung bình của $C$ theo $x$ khi mức sản xuất thay đổi
  \begin{enumerate}[label=(\roman*),leftmargin=1.6em,itemsep=0pt,topsep=1pt]
    \item từ $x = 100$ đến $x = 105$
    \item từ $x = 100$ đến $x = 101$
  \end{enumerate}
  \item Tìm tốc độ thay đổi tức thời của $C$ theo $x$ khi $x = 100$. (Đại lượng này được gọi là \textit{chi phí cận biên}. Ý nghĩa của nó sẽ được giải thích trong Mục 3.7.)
\end{enumerate}

\vspace{5pt}

\textbf{50.}\enskip Gọi $H(t)$ là chi phí hàng ngày (tính bằng đô la) để sưởi ấm một tòa nhà văn phòng khi nhiệt độ ngoài trời là $t$ độ Fahrenheit.
\begin{enumerate}[label=(\alph*),leftmargin=1.4em,itemsep=1pt,topsep=2pt]
  \item Ý nghĩa của $H'(58)$ là gì? Đơn vị của nó là gì?
  \item Bạn dự đoán $H'(58)$ mang dấu dương hay âm? Hãy giải thích.
\end{enumerate}

\vspace{5pt}

\textbf{51.}\enskip Chi phí để khai thác $x$ ounce vàng từ một mỏ vàng mới là $C = f(x)$ đô la.
\begin{enumerate}[label=(\alph*),leftmargin=1.4em,itemsep=1pt,topsep=2pt]
  \item Ý nghĩa của đạo hàm $f'(x)$ là gì? Đơn vị của nó là gì?
  \item Đẳng thức $f'(800) = 17$ có ý nghĩa gì?
  \item Bạn nghĩ giá trị của $f'(x)$ sẽ tăng hay giảm trong ngắn hạn? Còn trong dài hạn thì sao? Hãy giải thích.
\end{enumerate}

\vspace{5pt}

\textbf{52.}\enskip Lượng cà phê xay thượng hạng (tính bằng pound) mà một công ty cà phê bán được ở mức giá $p$ đô la mỗi pound là $Q = f(p)$.

  \item $f'(8)$ mang dấu dương hay âm? Hãy giải thích.
\end{enumerate}
