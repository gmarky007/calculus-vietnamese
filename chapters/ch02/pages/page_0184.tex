        \item Viết biểu thức cho hệ số góc của tiếp tuyến tại $P$.
      \end{enumerate}

    \item[\llap{\graphicon\hspace{0.35em}}\textbf{2.}] Vẽ đồ thị đường cong $y = e^x$ trong các hình chữ nhật quan sát $[-1, 1]$ theo $[0, 2]$, $[-0.5, 0.5]$ theo $[0.5, 1.5]$, và $[-0.1, 0.1]$ theo $[0.9, 1.1]$. Em nhận thấy điều gì về đường cong khi phóng to dần về phía điểm $(0, 1)$?

    \item 
      \begin{enumerate}[label=(\alph*), leftmargin=1.3em, labelsep=0.35em, itemsep=1.5pt, topsep=0pt]
        \item Tìm hệ số góc của tiếp tuyến với parabol $y = x^2 + 3x$ tại điểm $(-1, -2)$
          \vspace{1pt}
          \newline\phantom{}\hspace{0.4em}(i) sử dụng Định nghĩa 1 \quad (ii) sử dụng Phương trình 2
        \item Tìm phương trình tiếp tuyến ở câu (a).
        \item[\llap{\graphicon\hspace{0.35em}}(c)] Vẽ đồ thị parabol và tiếp tuyến. Để kiểm tra lại kết quả, hãy phóng to dần về phía điểm $(-1, -2)$ cho đến khi parabol và tiếp tuyến hầu như không thể phân biệt được.
      \end{enumerate}

    \item 
      \begin{enumerate}[label=(\alph*), leftmargin=1.3em, labelsep=0.35em, itemsep=1.5pt, topsep=0pt]
        \item Tìm hệ số góc của tiếp tuyến với đường cong $y = x^3 + 1$ tại điểm $(1, 2)$
          \vspace{1pt}
          \newline\phantom{}\hspace{0.4em}(i) sử dụng Định nghĩa 1 \quad (ii) sử dụng Phương trình 2
        \item Tìm phương trình tiếp tuyến ở câu (a).
        \item[\llap{\graphicon\hspace{0.35em}}(c)] Vẽ đồ thị đường cong và tiếp tuyến trong các hình chữ nhật quan sát thu nhỏ dần có tâm tại $(1, 2)$ cho đến khi đường cong và đường thẳng dường như trùng khít nhau.
      \end{enumerate}

  \begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=1.4em, labelsep=0.4em, itemsep=0.6em, topsep=0pt, parsep=0pt, start=12]
    \item Nếu một hòn đá được ném thẳng đứng lên trên ở Sao Hỏa với vận tốc $10\text{ m/s}$, độ cao của nó (tính bằng mét) sau $t$ giây được cho bởi $H = 10t - 1.86t^2$.
      \begin{enumerate}[label=(\alph*), leftmargin=1.3em, labelsep=0.35em, itemsep=1.5pt, topsep=1.5pt]
        \item Tìm vận tốc của hòn đá sau một giây.
        \item Tìm vận tốc của hòn đá khi $t = a$.
        \item Khi nào thì hòn đá chạm bề mặt?
        \item Hòn đá chạm bề mặt với vận tốc bằng bao nhiêu?
      \end{enumerate}

    \item Độ dịch chuyển (tính bằng mét) của một chất điểm chuyển động trên một đường thẳng được cho bởi phương trình chuyển động $s = 1/t^2$, trong đó $t$ được đo bằng giây. Tìm vận tốc của chất điểm tại các thời điểm $t = a$, $t = 1$, $t = 2$, và $t = 3$.

    \item Độ dịch chuyển (tính bằng feet) của một chất điểm chuyển động trên một đường thẳng được cho bởi $s = \frac{1}{2}t^2 - 6t + 23$, trong đó $t$ được đo bằng giây.
      \begin{enumerate}[label=(\alph*), leftmargin=1.3em, labelsep=0.35em, itemsep=1.5pt, topsep=1.5pt]
        \item Tìm vận tốc trung bình trên mỗi khoảng thời gian sau:
          \vspace{1.5pt}
          \newline\phantom{}\hspace{0.4em}(i) $[4, 8]$ \qquad (ii) $[6, 8]$ \qquad (iii) $[8, 10]$ \qquad (iv) $[8, 12]$
        \item Tìm vận tốc tức thời khi $t = 8$.
        \item Vẽ đồ thị của $s$ theo biến $t$ và vẽ các cát tuyến có hệ số góc là các vận tốc trung bình ở câu (a). Sau đó vẽ tiếp tuyến có hệ số góc là vận tốc tức thời ở câu (b).
      \end{enumerate}

    \item 
      \begin{enumerate}[label=(\alph*), leftmargin=1.3em, labelsep=0.35em, itemsep=1.5pt, topsep=0pt]
        \item Một chất điểm bắt đầu chuyển động sang phải dọc theo một đường thẳng nằm ngang; đồ thị hàm vị trí của nó được cho trong hình vẽ. Khi nào chất điểm chuyển động sang phải? Chuyển động sang trái? Đứng yên?
