\textbf{\color{stewartred}63.}\enspace \textbf{Đạo hàm của hàm số chẵn và hàm số lẻ}\enspace Nhắc lại rằng một hàm số $f$ được gọi là \textit{chẵn} nếu $f(-x) = f(x)$ với mọi $x$ thuộc tập xác định của nó và \textit{lẻ} nếu $f(-x) = -f(x)$ với mọi $x$ như vậy. Hãy chứng minh mỗi khẳng định sau:\par
\hspace*{1.6em}\textbf{(a)}\enspace Đạo hàm của một hàm số chẵn là một hàm số lẻ.\par
\hspace*{1.6em}\textbf{(b)}\enspace Đạo hàm của một hàm số lẻ là một hàm số chẵn.

\vspace{1pt}
\textbf{\color{stewartcyan}64--65\quad Đạo hàm bên trái và bên phải}\enspace \textit{Đạo hàm bên trái} và \textit{đạo hàm bên phải} của $f$ tại $a$ được định nghĩa lần lượt bởi
\begin{align*}
f'_-(a) &= \lim_{h \to 0^-} \frac{f(a+h) - f(a)}{h} \\[2pt]
\text{và}\qquad f'_+(a) &= \lim_{h \to 0^+} \frac{f(a+h) - f(a)}{h}
\end{align*}
nếu các giới hạn này tồn tại. Khi đó $f'(a)$ tồn tại khi và chỉ khi các đạo hàm một phía này cùng tồn tại và bằng nhau.
