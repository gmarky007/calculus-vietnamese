\textbf{29.} Cho
\[

\setlength{\belowdisplayskip}{2pt}

\sqrt{-x} & \text{nếu } x < 0 \\

(x - 3)^2 & \text{nếu } x > 3
\end{cases}

    (iv) $\lim\limits_{x\to 3^-} f(x)$ & (v) $\lim\limits_{x\to 3^+} f(x)$ & (vi) $\lim\limits_{x\to 3} f(x)$

\textbf{30.} Cho
\[
\setlength{\abovedisplayskip}{2pt}
\setlength{\belowdisplayskip}{2pt}

2 - x & \text{nếu } 2 < x \le 3 \\
x - 4 & \text{nếu } 3 < x < 4 \\
