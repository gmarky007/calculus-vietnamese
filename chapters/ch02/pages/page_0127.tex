\prob{7} Đối với hàm số $g$ có đồ thị như hình vẽ, hãy tìm một số $a$ thỏa mãn mô tả đã cho.
\begin{enumerate}[label=\textbf{(\alph*)}, leftmargin=1.6em, itemsep=1pt, topsep=1pt]
  \item $\displaystyle \lim_{x \to a} g(x)$ không tồn tại nhưng $g(a)$ được xác định.
  \item $\displaystyle \lim_{x \to a} g(x)$ tồn tại nhưng $g(a)$ không được xác định.
  \item $\displaystyle \lim_{x \to a^-} g(x)$ và $\displaystyle \lim_{x \to a^+} g(x)$ đều tồn tại nhưng $\displaystyle \lim_{x \to a} g(x)$ không tồn tại.
  \item $\displaystyle \lim_{x \to a^+} g(x) = g(a)$ nhưng $\displaystyle \lim_{x \to a^-} g(x) \neq g(a)$.
\end{enumerate}
