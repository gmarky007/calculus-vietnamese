% 3. Nội dung lý thuyết và ví dụ trong cột chính
\noindent
\hspace*{0.29\textwidth}%
\begin{minipage}[t]{0.71\textwidth}
  % Tiêu đề tiểu mục
  {\color{stewartcyan}\rule{1.1ex}{1.1ex}}\hspace{0.5em}\textbf{\textsf{\large Hàm đạo hàm}}
  \par\vspace{0.7em}

  Trong mục trước, chúng ta đã xét đạo hàm của một hàm số $f$ tại một số cố định $a$:
  \par\vspace{0.8em}

  % Công thức (1)
  \noindent
  \makebox[0pt][l]{\eqntag{1}}\hfill
  $\displaystyle f'(a) = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}$
  \hfill\null
  \par\vspace{0.8em}

  Ở đây chúng ta thay đổi quan điểm và cho số $a$ biến thiên. Nếu ta thay $a$ trong Phương trình~1 bằng một biến số $x$, ta nhận được
  \par\vspace{0.8em}

  % Công thức (2) đóng khung viền đỏ
  \noindent
  \makebox[0pt][l]{\eqntag{2}}\hfill
  \setlength{\fboxrule}{0.8pt}\setlength{\fboxsep}{6.5pt}%
  \fcolorbox{stewartred}{white}{%
    \hspace{2.2em}$\displaystyle f'(x) = \lim_{h \to 0} \frac{f(x+h) - f(x)}{h}$\hspace{2.2em}%
  }\hfill\null
  \par\vspace{1.1em}

  Cho trước bất kỳ số $x$ nào mà giới hạn này tồn tại, ta gán cho $x$ số $f'(x)$. Do đó, ta có thể xem $f'$ là một hàm số mới, được gọi là \textbf{đạo hàm của $\boldsymbol{f}$} và được xác định bởi Phương trình~2. Ta biết rằng giá trị của $f'$ tại $x$, tức $f'(x)$, có thể được giải thích về mặt hình học là hệ số góc của tiếp tuyến với đồ thị của $f$ tại điểm $(x, f(x))$.
  \par\vspace{0.8em}

  Hàm số $f'$ được gọi là đạo hàm của $f$ bởi vì nó được “dẫn xuất” từ $f$ thông qua phép toán lấy giới hạn trong Phương trình~2. Tập xác định của $f'$ là tập hợp $\{x \mid f'(x)\text{ tồn tại}\}$ và có thể nhỏ hơn tập xác định của $f$.
  \par\vspace{1.3em}

  % Ví dụ 1
  \textbf{\textsf{\color{stewartcyan}VÍ DỤ 1}}\hspace{1em}Đồ thị của một hàm số $f$ được cho trong Hình~1. Hãy dùng nó để phác thảo đồ thị của hàm đạo hàm $f'$.
\end{minipage}
