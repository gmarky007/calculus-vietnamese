  \vspace{0pt}
  \noindent\textbf{\textsf{\color{stewartcyan}VÍ DỤ 6}}\hspace{0.8em}Tìm phương trình tiếp tuyến của parabol $y = x^2 - 8x + 9$ tại điểm $(3, -6)$.

  \vspace{0.4em}
  \noindent\textbf{\textsf{\color{stewartcyan}LỜI GIẢI}}\hspace{0.8em}Từ Ví dụ 4(b), ta biết rằng đạo hàm của $f(x) = x^2 - 8x + 9$ tại số $a$ là $f'(a) = 2a - 8$. Do đó, hệ số góc của tiếp tuyến tại $(3, -6)$ là $f'(3) = 2(3) - 8 = -2$. Vì vậy, phương trình của tiếp tuyến, được thể hiện trong Hình 7, là
  \begin{equation*}
    y - (-6) = (-2)(x - 3) \qquad\text{hay}\qquad y = -2x \tag*{\color{stewartcyan}\blacksquare}
  \end{equation*}

  \vspace{0.6em}
  \noindent{\color{stewartcyan}\rule{1.1ex}{1.1ex}}\hspace{0.5em}\textbf{\textsf{\large Tốc độ thay đổi}}

  \vspace{0.3em}
  \noindent Giả sử $y$ là một đại lượng phụ thuộc vào một đại lượng khác $x$. Khi đó $y$ là một hàm số của $x$ và ta viết $y = f(x)$. Nếu $x$ thay đổi từ $x_1$ sang $x_2$, thì độ biến thiên của $x$ (còn gọi là \textbf{số gia} của $x$) là
  \[
    \Delta x = x_2 - x_1
  \]
  và độ biến thiên tương ứng của $y$ là
  \[
    \Delta y = f(x_2) - f(x_1)
  \]
  Tỉ số sai phân
  \[
    \frac{\Delta y}{\Delta x} = \frac{f(x_2) - f(x_1)}{x_2 - x_1}
  \]
  được gọi là \textbf{tốc độ thay đổi trung bình của $y$ theo $x$} trên đoạn $[x_1, x_2]$ và có thể được hiểu là hệ số góc của cát tuyến $PQ$ trong Hình 8.

  \begin{center}
    \vspace{0.1em}
    \includegraphics[width=0.82\linewidth]{images/p0181_fig2.png}
    \vspace{0.1em}
  \end{center}

  \noindent Tương tự như đối với vận tốc, chúng ta xét tốc độ thay đổi trung bình trên các khoảng ngày càng nhỏ hơn bằng cách cho $x_2$ tiến dần đến $x_1$ và do đó cho $\Delta x$ tiến dần đến $0$. Giới hạn của các tốc độ thay đổi trung bình này được gọi là \textbf{tốc độ thay đổi (tức thời) của $y$ theo $x$} tại $x = x_1$, và (như trong trường hợp vận tốc) được hiểu là hệ số góc của tiếp tuyến với đường cong $y = f(x)$ tại $P(x_1, f(x_1))$:

  \vspace{0.3em}
  \begin{tcolorbox}[
    colback=white,
    colframe=stewartred,
    boxrule=0.8pt,
    sharp corners,
    boxsep=0pt,
    left=8pt,
    right=8pt,
    top=6pt,
    bottom=6pt
  ]
    \noindent
    \parbox{0.06\linewidth}{%
      \tikz\node[draw=stewartred, line width=0.8pt, inner sep=2pt, font=\bfseries\sffamily\color{stewartred}] {6};%
    }%
    \parbox{0.92\linewidth}{%
      \centering
      $\displaystyle \text{tốc độ thay đổi tức thời} = \lim_{\Delta x \to 0} \frac{\Delta y}{\Delta x} = \lim_{x_2 \to x_1} \frac{f(x_2) - f(x_1)}{x_2 - x_1}$
    }
  \end{tcolorbox}

  \vspace{0.4em}
  \noindent Chúng ta nhận thấy giới hạn này chính là đạo hàm $f'(x_1)$.

  \vspace{0.2em}
  \noindent Chúng ta biết rằng một ý nghĩa của đạo hàm $f'(a)$ là hệ số góc của tiếp tuyến với đường cong $y = f(x)$ khi $x = a$. Bây giờ chúng ta có ý nghĩa thứ hai:

  \vspace{0.3em}
  \begin{tcolorbox}[
    colback=white,
    colframe=stewartred,
    boxrule=0.8pt,
    sharp corners,
    boxsep=0pt,
    left=10pt,
    right=10pt,
    top=8pt,
    bottom=8pt
  ]
    Đạo hàm $f'(a)$ là \textbf{tốc độ thay đổi tức thời của $y = f(x)$ theo $x$ khi $x = a$}.
  \end{tcolorbox}
