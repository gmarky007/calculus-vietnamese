    Cuối cùng, ta lưu ý rằng một giới hạn vô cực ở vô cực có thể được định nghĩa như sau. Minh họa hình học được cho trong Hình~19.

    \vspace{0.4em}
    \begin{tcolorbox}[
        colframe=stewartred,
        colback=white,
        boxrule=0.8pt,
        arc=0mm,
        left=8pt,
        right=8pt,
        top=6pt,
        bottom=6pt
    ]
        \noindent
        \colorbox{stewartred}{\color{white}\bfseries\textsf{9}}\enspace
        {\bfseries\color{stewartred}\textsf{Định nghĩa chính xác về giới hạn vô cực ở vô cực}}\enspace
        Cho $f$ là một hàm số xác định trên một khoảng $(a, \infty)$ nào đó. Khi đó
        \[
            \lim_{x \to \infty} f(x) = \infty
        \]
        có nghĩa là với mọi số dương $M$, tồn tại một số dương $N$ tương ứng sao cho
        \[
            \text{nếu } x > N \quad \text{thì } f(x) > M
        \]
    \end{tcolorbox}
    
    \vspace{0.3em}
    Các định nghĩa tương tự cũng áp dụng khi ký hiệu $\infty$ được thay bằng $-\infty$. (Xem Bài tập 80.)
\end{minipage}

\begin{minipage}[t]{0.48\textwidth}
    \vspace{0pt}
    \begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=1.5em, itemsep=0.6em]
        \item Hãy giải thích bằng lời của bạn ý nghĩa của mỗi biểu thức sau.\\[3pt]
        (a) $\displaystyle\lim_{x \to \infty} f(x) = 5$ \hfill (b) $\displaystyle\lim_{x \to -\infty} f(x) = 3$

        \item (a) Đồ thị của hàm số $y = f(x)$ có thể cắt một đường tiệm cận đứng không? Nó có thể cắt một đường tiệm cận ngang không? Hãy vẽ đồ thị phác thảo để minh họa.\\[2pt]
        (b) Đồ thị của $y = f(x)$ có thể có bao nhiêu đường tiệm cận ngang? Hãy vẽ phác thảo đồ thị để minh họa các khả năng.

        \item Cho hàm số $f$ có đồ thị như hình bên, hãy xác định các giá trị sau.\\[3pt]
        (a) $\displaystyle\lim_{x \to \infty} f(x)$ \hfill (b) $\displaystyle\lim_{x \to -\infty} f(x)$\\[3pt]
        (c) $\displaystyle\lim_{x \to 1} f(x)$ \hfill (d) $\displaystyle\lim_{x \to 3} f(x)$\\[3pt]
        (e) Phương trình của các đường tiệm cận
    \end{enumerate}

        \item Cho hàm số $g$ có đồ thị như hình trên, hãy xác định các giá trị sau.\\[3pt]
        (a) $\displaystyle\lim_{x \to \infty} g(x)$ \hfill (b) $\displaystyle\lim_{x \to -\infty} g(x)$\\[3pt]
        (c) $\displaystyle\lim_{x \to 0} g(x)$ \hfill (d) $\displaystyle\lim_{x \to 2^-} g(x)$\\[3pt]
        (e) $\displaystyle\lim_{x \to 2^+} g(x)$ \hfill (f) Phương trình các tiệm cận
    \end{enumerate}

    \textbf{Hãy phác thảo đồ thị của một ví dụ về hàm số $f$ thỏa mãn tất cả các điều kiện đã cho.}

    \vspace{0.4em}
    \begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=1.5em, start=5, itemsep=0.55em]
        \item $f(2) = 4, \quad f(-2) = -4, \quad \displaystyle\lim_{x \to -\infty} f(x) = 0, \quad \lim_{x \to \infty} f(x) = 2$

        \item $f(0) = 0, \quad \displaystyle\lim_{x \to 1^-} f(x) = \infty, \quad \lim_{x \to 1^+} f(x) = -\infty,$\\[2pt]
        $\displaystyle\lim_{x \to -\infty} f(x) = -2, \quad \lim_{x \to \infty} f(x) = -2$

        \item $\displaystyle\lim_{x \to 0} f(x) = \infty, \quad \lim_{x \to 3^-} f(x) = -\infty, \quad \lim_{x \to 3^+} f(x) = \infty,$\\[2pt]
        $\displaystyle\lim_{x \to -\infty} f(x) = 1, \quad \lim_{x \to \infty} f(x) = -1$

        \item $\displaystyle\lim_{x \to -\infty} f(x) = -\infty, \quad \lim_{x \to -2^-} f(x) = \infty, \quad \lim_{x \to -2^+} f(x) = -\infty,$\\[2pt]
        $\displaystyle\lim_{x \to 2} f(x) = \infty, \quad \lim_{x \to \infty} f(x) = \infty$

        \item $f(0) = 0, \quad \displaystyle\lim_{x \to 1} f(x) = -\infty, \quad \lim_{x \to \infty} f(x) = -\infty, \quad f \text{ là hàm số lẻ}$

        \item $\displaystyle\lim_{x \to -\infty} f(x) = -1, \quad \lim_{x \to 0^-} f(x) = \infty, \quad \lim_{x \to 0^+} f(x) = -\infty,$\\[2pt]
        $\displaystyle\lim_{x \to 3^-} f(x) = 1, \quad f(3) = 4, \quad \lim_{x \to 3^+} f(x) = 4, \quad \lim_{x \to \infty} f(x) = 1$
    \end{enumerate}
