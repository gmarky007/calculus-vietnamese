\noindent{\small\textbf{\textsf{HÌNH 11}}}\\[1pt]
{\small $\lim\limits_{x \to a} f(x) = -\infty$}

\switchcolumn
% --- CỘT NỘI DUNG CHÍNH ---
\noindent Một loại giới hạn tương tự, đối với các hàm số trở nên âm rất lớn khi $x$ tiến gần tới $a$, được nêu trong Định nghĩa 5 và được minh họa trong Hình 11.

\vspace{0.7em}

\begin{tcolorbox}[
    enhanced,
    colback=white,
    colframe=stewartred,
    arc=0mm,
    boxrule=0.8pt,
    left=8pt, right=8pt, top=6pt, bottom=6pt
]
\noindent\fboxrule=0.8pt\fboxsep=2.5pt\fcolorbox{stewartred}{white}{\bfseries\color{stewartred}5}\hspace{0.5em}{\bfseries\color{stewartred} Định nghĩa}\hspace{0.8em}Cho $f$ là một hàm số xác định trên cả hai phía của $a$, có thể ngoại trừ tại chính $a$. Khi đó
\[
    \lim_{x \to a} f(x) = -\infty
\]
có nghĩa là các giá trị của $f(x)$ có thể trở nên âm lớn tùy ý bằng cách lấy $x$ đủ gần $a$, nhưng không bằng $a$.
\end{tcolorbox}

\vspace{0.7em}

\noindent Ký hiệu $\lim_{x \to a} f(x) = -\infty$ có thể đọc là “giới hạn của $f(x)$, khi $x$ tiến tới $a$, bằng âm vô cùng” hoặc “$f(x)$ giảm không giới hạn khi $x$ tiến tới $a$”. Chẳng hạn, ta có
\[
    \lim_{x \to 0} \left( -\frac{1}{x^2} \right) = -\infty
\]

\noindent Các định nghĩa tương tự có thể được đưa ra cho các giới hạn vô cực một phía
\[
    \lim_{x \to a^-} f(x) = \infty \qquad\qquad \lim_{x \to a^+} f(x) = \infty
\]
\[
    \lim_{x \to a^-} f(x) = -\infty \qquad\qquad \lim_{x \to a^+} f(x) = -\infty
\]
\noindent lưu ý rằng $x \to a^-$ có nghĩa là chúng ta chỉ xét các giá trị của $x$ nhỏ hơn $a$, và tương tự $x \to a^+$ có nghĩa là chúng ta chỉ xét $x > a$. Các minh họa cho bốn trường hợp này được thể hiện trong Hình 12.

\end{paracol}

\noindent\fboxrule=0.8pt\fboxsep=2.5pt\fcolorbox{stewartred}{white}{\bfseries\color{stewartred}6}\hspace{0.5em}{\bfseries\color{stewartred} Định nghĩa}\hspace{0.8em}Đường thẳng đứng $x = a$ được gọi là một \textbf{tiệm cận đứng} của đường cong $y = f(x)$ nếu có ít nhất một trong các khẳng định sau là đúng:
\[
    \lim_{x \to a} f(x) = \infty \qquad\qquad \lim_{x \to a^-} f(x) = \infty \qquad\qquad \lim_{x \to a^+} f(x) = \infty
\]
\[
    \lim_{x \to a} f(x) = -\infty \qquad\qquad \lim_{x \to a^-} f(x) = -\infty \qquad\qquad \lim_{x \to a^+} f(x) = -\infty
\]
\end{tcolorbox}

\vspace{0.8em}

\noindent Chẳng hạn, trục $y$ là một tiệm cận đứng của đường cong $y = 1/x^2$ vì $\lim_{x \to 0} (1/x^2) = \infty$. Trong Hình 12, đường thẳng $x = a$ là tiệm cận đứng trong mỗi trường hợp trong bốn trường hợp được thể hiện. Nhìn chung, hiểu biết về tiệm cận đứng rất hữu ích trong việc phác họa đồ thị.
