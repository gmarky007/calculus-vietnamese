\begin{definitionbox}
\noindent\textbf{\textsf{\color{stewartred}Các quy tắc giới hạn}}\quad Giả sử $c$ là một hằng số và các giới hạn
\[
\lim_{x \to a} f(x) \qquad \text{và} \qquad \lim_{x \to a} g(x)
\]
tồn tại. Khi đó
\begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=1.8em, itemsep=4pt, topsep=3pt, parsep=0pt, partopsep=0pt]
\item $\displaystyle \lim_{x \to a} [f(x) + g(x)] = \lim_{x \to a} f(x) + \lim_{x \to a} g(x)$
\item $\displaystyle \lim_{x \to a} [f(x) - g(x)] = \lim_{x \to a} f(x) - \lim_{x \to a} g(x)$
\item $\displaystyle \lim_{x \to a} [c f(x)] = c \lim_{x \to a} f(x)$
\item $\displaystyle \lim_{x \to a} [f(x) g(x)] = \lim_{x \to a} f(x) \cdot \lim_{x \to a} g(x)$
\item $\displaystyle \lim_{x \to a} \frac{f(x)}{g(x)} = \frac{\displaystyle\lim_{x \to a} f(x)}{\displaystyle\lim_{x \to a} g(x)} \qquad \text{nếu } \lim_{x \to a} g(x) \neq 0$
\end{enumerate}
\end{definitionbox}

Thật dễ để tin rằng các tính chất này là đúng. Chẳng hạn, nếu $f(x)$ gần với $L$ và $g(x)$ gần với $M$, thì hoàn toàn hợp lý khi kết luận rằng $f(x) + g(x)$ gần với $L + M$. Điều này cho chúng ta một cơ sở trực giác để tin rằng Quy tắc 1 là đúng. Trong Mục 2.4, chúng ta sẽ đưa ra định nghĩa chính xác về giới hạn và dùng nó để chứng minh quy tắc này. Các chứng minh cho các quy tắc còn lại được trình bày trong Phụ lục F.

\noindent\textbf{\textsf{\color{stewartcyan}VÍ DỤ 1}}\quad Sử dụng các Quy tắc giới hạn và đồ thị của $f$ và $g$ trong Hình 1 để tính các giới hạn sau, nếu chúng tồn tại.
\vspace{6pt}

\noindent
\textbf{(a)}\enskip $\displaystyle \lim_{x \to -2} [f(x) + 5g(x)]$ \hfill
\textbf{(b)}\enskip $\displaystyle \lim_{x \to 1} [f(x)g(x)]$ \hfill
\textbf{(c)}\enskip $\displaystyle \lim_{x \to 2} \frac{f(x)}{g(x)}$
\vspace{8pt}

\noindent\textbf{\textsf{\color{stewartcyan}LỜI GIẢI}}\quad
\textbf{(a)}\enskip Từ đồ thị của $f$ và $g$, ta thấy rằng
\[
\lim_{x \to -2} f(x) = 1 \qquad \text{và} \qquad \lim_{x \to -2} g(x) = -1
\]
Do đó ta có
\begin{align*}
\lim_{x \to -2} [f(x) + 5g(x)] &= \lim_{x \to -2} f(x) + \lim_{x \to -2} [5g(x)] && \text{\textcolor{stewartcyan}{(theo Quy tắc giới hạn 1)}} \\
&= \lim_{x \to -2} f(x) + 5\lim_{x \to -2} g(x) && \text{\textcolor{stewartcyan}{(theo Quy tắc giới hạn 3)}} \\
&= 1 + 5(-1) = -4
\end{align*}
