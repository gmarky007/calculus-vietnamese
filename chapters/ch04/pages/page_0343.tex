\noindent\makebox[0pt][r]{\casicon\hspace{4pt}}\textbf{81.} Một \textit{đường cong đáp ứng thuốc} mô tả nồng độ của thuốc trong máu sau khi dùng thuốc. Một hàm tăng vọt $S(t) = At^p e^{-kt}$ thường được dùng để mô hình hóa đường cong đáp ứng, thể hiện sự gia tăng đột ngột ban đầu về nồng độ thuốc rồi sau đó giảm dần từ từ hơn. Nếu đối với một loại thuốc cụ thể, $A = 0{,}01$, $p = 4$, $k = 0{,}07$, và $t$ được đo bằng phút, hãy ước tính các thời điểm tương ứng với các điểm uốn và giải thích ý nghĩa của chúng. Sau đó, vẽ đồ thị của đường cong đáp ứng thuốc.

\textbf{82. Các hàm mật độ chuẩn}\quad Họ các đường cong hình chuông
\[
y = \frac{1}{\sigma \sqrt{2\pi}}\, e^{-(x-\mu)^2/(2\sigma^2)}
\]
xuất hiện trong lý thuyết xác suất và thống kê, nơi nó được gọi là \textit{hàm mật độ chuẩn}. Hằng số $\mu$ được gọi là \textit{giá trị trung bình} và hằng số dương $\sigma$ được gọi là \textit{độ lệch chuẩn}. Để cho đơn giản, ta thu phóng hàm số để loại bỏ hệ số $1/(\sigma \sqrt{2\pi})$ và xét trường hợp đặc biệt $\mu = 0$. Vì vậy, ta khảo sát hàm số
\[
f(x) = e^{-x^2/(2\sigma^2)}
\]
\begin{enumerate}[label=(\alph*), itemsep=0.5pt, topsep=1.5pt, parsep=0pt, leftmargin=1.5em]
    \item Tìm các đường tiệm cận, giá trị lớn nhất và các điểm uốn của $f$.
\end{enumerate}

\textbf{88.}
\begin{enumerate}[label=(\alph*), itemsep=0.5pt, topsep=1.5pt, parsep=0pt, leftmargin=1.5em]
    \item Nếu $f$ và $g$ có bề lõm hướng lên trên trên một khoảng $I$, hãy chứng minh rằng $f + g$ cũng có bề lõm hướng lên trên trên $I$.
    \item Nếu $f$ dương và có bề lõm hướng lên trên trên $I$, hãy chứng minh rằng hàm số $g(x) = [f(x)]^2$ cũng có bề lõm hướng lên trên trên $I$.
\end{enumerate}

\vspace{1.8mm}
\textbf{89.}
\begin{enumerate}[label=(\alph*), itemsep=0.5pt, topsep=1.5pt, parsep=0pt, leftmargin=1.5em]
    \item Nếu $f$ và $g$ là các hàm số dương, đồng biến và có bề lõm hướng lên trên trên một khoảng $I$, hãy chứng minh rằng hàm tích $fg$ cũng có bề lõm hướng lên trên trên $I$.
    \item Chứng minh rằng kết luận ở phần (a) vẫn đúng nếu cả $f$ và $g$ đều nghịch biến.
    \item Giả sử $f$ đồng biến và $g$ nghịch biến. Bằng cách đưa ra ba ví dụ, hãy chỉ ra rằng $fg$ có thể có bề lõm hướng lên trên, hướng xuống dưới hoặc tuyến tính. Tại sao lập luận ở các phần (a) và (b) không áp dụng được trong trường hợp này?
\end{enumerate}

\noindent\makebox[0pt][r]{\casicon\hspace{4pt}}\textbf{92.} Với những giá trị nào của $c$ thì đa thức
\[
P(x) = x^4 + cx^3 + x^2
\]

\textbf{93.} Chứng minh rằng nếu $(c, f(c))$ là một điểm uốn của đồ thị hàm số $f$ và $f''$ tồn tại trên một khoảng mở chứa $c$, thì $f''(c) = 0$. [\textit{Gợi ý:} Áp dụng Kiểm tra Đạo hàm cấp một và Định lý Fermat cho hàm số $g = f'$.]

\vspace{2mm}
\textbf{94.} Chứng minh rằng nếu $f(x) = x^4$, thì $f''(0) = 0$, nhưng $(0, 0)$ không phải là một điểm uốn của đồ thị hàm số $f$.
