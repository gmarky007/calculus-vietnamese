    \textbf{B. Giao điểm với các trục}\quad Cả giao điểm với trục $x$ và trục $y$ đều là $0$.

    \textbf{C. Tính đối xứng}\quad Không có.

    \textbf{D. Tiệm cận}\quad Vì
    \[
    \lim_{x \to \infty} \frac{x^2}{\sqrt{x+1}} = \infty
    \]
    nên không có tiệm cận ngang. Vì $\sqrt{x+1} \to 0$ khi $x \to -1^+$ và $f(x)$ luôn dương, ta có
    \[
    \lim_{x \to -1^+} \frac{x^2}{\sqrt{x+1}} = \infty
    \]
    và do đó đường thẳng $x = -1$ là một tiệm cận đứng.

    \textbf{E. Các khoảng tăng hoặc giảm}
    \[
    f'(x) = \frac{\sqrt{x+1}(2x) - x^2 \cdot 1/(2\sqrt{x+1})}{x+1} = \frac{3x^2 + 4x}{2(x+1)^{3/2}} = \frac{x(3x+4)}{2(x+1)^{3/2}}
    \]
    Ta thấy rằng $f'(x) = 0$ khi $x = 0$ (lưu ý rằng $-\frac{4}{3}$ không thuộc tập xác định của $f$), do đó số tới hạn duy nhất là $0$. Vì $f'(x) < 0$ khi $-1 < x < 0$ và $f'(x) > 0$ khi $x > 0$, nên $f$ nghịch biến trên $(-1, 0)$ và đồng biến trên $(0, \infty)$.

    \textbf{F. Giá trị cực đại hoặc cực tiểu địa phương}\quad Vì $f'(0) = 0$ và $f'$ đổi dấu từ âm sang dương tại $0$, nên theo Kiểm định đạo hàm cấp một, $f(0) = 0$ là một giá trị cực tiểu địa phương (và cũng là cực tiểu tuyệt đối).

    \textbf{G. Tính lồi lõm và điểm uốn}
    \[
    f''(x) = \frac{2(x+1)^{3/2}(6x+4) - (3x^2 + 4x)3(x+1)^{1/2}}{4(x+1)^3} = \frac{3x^2 + 8x + 8}{4(x+1)^{5/2}}
    \]
    Lưu ý rằng mẫu số luôn luôn dương. Tử số là tam thức bậc hai $3x^2 + 8x + 8$, luôn dương vì có biệt thức $b^2 - 4ac = -32$ âm, và hệ số của $x^2$ là dương. Do đó $f''(x) > 0$ với mọi $x$ thuộc tập xác định của $f$, điều này có nghĩa là $f$ lõm lên trên $(-1, \infty)$ và đồ thị không có điểm uốn.

    \textbf{H. Vẽ đồ thị}\quad Đường cong được phác họa như trong Hình 7.\hfill{\color{stewartcyan}\rule{1.1ex}{1.1ex}}

    \noindent\textbf{\textsf{\color{stewartcyan}VÍ DỤ 3}}\quad Phác thảo đồ thị của hàm số $f(x) = xe^x$.

    \textbf{A. Tập xác định}\quad Tập xác định là $\mathbb{R}$.

    \textbf{B. Giao điểm với các trục}\quad Cả giao điểm với trục $x$ và trục $y$ đều là $0$.

    \textbf{C. Tính đối xứng}\quad Không có.

    \textbf{D. Tiệm cận}\quad Vì cả $x$ và $e^x$ đều trở nên rất lớn khi $x \to \infty$, ta có $\lim_{x \to \infty} xe^x = \infty$. Tuy nhiên, khi $x \to -\infty$, $e^x \to 0$ và do đó ta có một dạng tích vô định đòi hỏi phải áp dụng quy tắc l'Hospital:
    \[
    \lim_{x \to -\infty} xe^x = \lim_{x \to -\infty} \frac{x}{e^{-x}} = \lim_{x \to -\infty} \frac{1}{-e^{-x}} = \lim_{x \to -\infty} (-e^x) = 0
    \]
    Do đó trục $x$ là một tiệm cận ngang.
