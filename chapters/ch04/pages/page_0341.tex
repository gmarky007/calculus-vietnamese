    {\color{stewartcyan}\textbf{34--41}}\enspace Phác họa đồ thị của một hàm số thỏa mãn tất cả các điều kiện đã cho.
    \vspace{3pt}

    \textbf{34.}\enspace \textbf{(a)} $f'(x) < 0$ và $f''(x) < 0$ với mọi $x$\\
    \phantom{\textbf{34.}\enspace}\textbf{(b)} $f'(x) > 0$ và $f''(x) > 0$ với mọi $x$\\[3pt]
    \textbf{35.}\enspace \textbf{(a)} $f'(x) > 0$ và $f''(x) < 0$ với mọi $x$\\
    \phantom{\textbf{35.}\enspace}\textbf{(b)} $f'(x) < 0$ và $f''(x) > 0$ với mọi $x$\\[3pt]
    \textbf{36.}\enspace Tiệm cận đứng $x = 0, \quad f'(x) > 0$ nếu $x < -2$,\\
    \phantom{\textbf{36.}\enspace}$f'(x) < 0$ nếu $x > -2 \; (x \neq 0)$,\\
    \phantom{\textbf{36.}\enspace}$f''(x) < 0$ nếu $x < 0, \quad f''(x) > 0$ nếu $x > 0$\\[3pt]
    \textbf{37.}\enspace $f'(0) = f'(2) = f'(4) = 0$,\\
    \phantom{\textbf{37.}\enspace}$f'(x) > 0$ nếu $x < 0$ hoặc $2 < x < 4$,\\
    \phantom{\textbf{37.}\enspace}$f'(x) < 0$ nếu $0 < x < 2$ hoặc $x > 4$,\\
    \phantom{\textbf{37.}\enspace}$f''(x) > 0$ nếu $1 < x < 3, \quad f''(x) < 0$ nếu $x < 1$ hoặc $x > 3$\\[3pt]
    \textbf{38.}\enspace $f'(x) > 0$ với mọi $x \neq 1, \quad$ tiệm cận đứng $x = 1$,\\
    \phantom{\textbf{38.}\enspace}$f''(x) > 0$ nếu $x < 1$ hoặc $x > 3, \quad f''(x) < 0$ nếu $1 < x < 3$\\[3pt]
    \textbf{39.}\enspace $f'(5) = 0, \quad f'(x) < 0$ khi $x < 5$,\\
    \phantom{\textbf{39.}\enspace}$f'(x) > 0$ khi $x > 5, \quad f''(2) = 0, \quad f''(8) = 0$,\\
    \phantom{\textbf{39.}\enspace}$f''(x) < 0$ khi $x < 2$ hoặc $x > 8$,\\
    \phantom{\textbf{39.}\enspace}$f''(x) > 0$ khi $2 < x < 8, \quad \lim\limits_{x \to \infty} f(x) = 3, \quad \lim\limits_{x \to -\infty} f(x) = 3$\\[3pt]
    \textbf{40.}\enspace $f'(0) = f'(4) = 0, \quad f'(x) = 1$ nếu $x < -1$,\\
    \phantom{\textbf{40.}\enspace}$f'(x) > 0$ nếu $0 < x < 2$,\\
    \phantom{\textbf{40.}\enspace}$f'(x) < 0$ nếu $-1 < x < 0$ hoặc $2 < x < 4$ hoặc $x > 4$,\\
    \phantom{\textbf{40.}\enspace}$\lim\limits_{x \to 2^-} f'(x) = \infty, \quad \lim\limits_{x \to 2^+} f'(x) = -\infty$,\\
    \phantom{\textbf{40.}\enspace}$f''(x) > 0$ nếu $-1 < x < 2$ hoặc $2 < x < 4$,\\
    \phantom{\textbf{40.}\enspace}$f''(x) < 0$ nếu $x > 4$\\[3pt]
    \textbf{41.}\enspace $f'(x) > 0$ nếu $x \neq 2, \quad f''(x) > 0$ nếu $x < 2$,\\
    \phantom{\textbf{41.}\enspace}$f''(x) < 0$ nếu $x > 2, \quad f$ có điểm uốn $(2, 5)$,\\
    \phantom{\textbf{41.}\enspace}$\lim\limits_{x \to \infty} f(x) = 8, \quad \lim\limits_{x \to -\infty} f(x) = 0$

    \exercisedivider

    \textbf{42.}\enspace Đồ thị của hàm số $y = f(x)$ được cho trong hình bên dưới. Tại (những) điểm nào thì các khẳng định sau là đúng?\\[3pt]
    \phantom{\textbf{42.}\enspace}\textbf{(a)} $\dfrac{dy}{dx}$ và $\dfrac{d^2y}{dx^2}$ đều mang giá trị dương.\\[4pt]
    \phantom{\textbf{42.}\enspace}\textbf{(b)} $\dfrac{dy}{dx}$ và $\dfrac{d^2y}{dx^2}$ đều mang giá trị âm.\\[4pt]
    \phantom{\textbf{42.}\enspace}\textbf{(c)} $\dfrac{dy}{dx}$ âm nhưng $\dfrac{d^2y}{dx^2}$ dương.
